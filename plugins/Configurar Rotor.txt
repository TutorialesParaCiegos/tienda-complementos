context = activity or service
local prefs = PreferenceManager.getDefaultSharedPreferences(service)
function getRotorTypes()
local rotorSet = prefs.getStringSet("view_type_move", nil)
local types = {}
if rotorSet then
local iterator = rotorSet.iterator()
while iterator.hasNext() do
local typeStr = iterator.next()
local typeNum = tonumber(typeStr)
if typeNum then
types[#types + 1] = typeNum
end
end
else
types = {0, 2, 14, 3, 15, 16, 5, 20, 21}
end
return types
end
function saveRotorTypes(types)
table.sort(types)
local typesSet = HashSet()
for _, typeNum in ipairs(types) do
typesSet.add(tostring(typeNum))
end
prefs.edit().putStringSet("view_type_move", typesSet).commit()
end

function showRotorConfigDialog()
local activeTypes = getRotorTypes()
local typesChecked = {}
for i = 0, 23 do
typesChecked[i + 1] = false
for _, typeNum in ipairs(activeTypes) do
if typeNum == i then
typesChecked[i + 1] = true
break
end
end
end

local configDlg = LuaDialog(service)
local configView = loadlayout(configLayout)
configDlg.setView(configView)
configDlg.setCancelable(false)
configDlg.setCanceledOnTouchOutside(false)
local checkboxes = {}
for i = 0, 23 do
local checkbox = CheckBox(service)
checkbox.setText(rotorNames[i])
checkbox.setTextSize(18)
checkbox.setChecked(typesChecked[i + 1])
checkbox.setPadding(8, 12, 8, 12)
checkbox.setTextColor(0xFFFFFFFF)
typesContainer.addView(checkbox)
checkboxes[i + 1] = checkbox
end
local function actualizarBotonToggle()
local hayAlgunaMarcada = false
for i = 1, 24 do
if checkboxes[i] and checkboxes[i].isChecked() then
hayAlgunaMarcada = true
break
end
end
if hayAlgunaMarcada then
btnToggleAll.setText("Desmarcar todo")
else
btnToggleAll.setText("Marcar todo")
end
end
for i = 1, 24 do
checkboxes[i].setOnClickListener(function()
actualizarBotonToggle()
end)
end
actualizarBotonToggle()
btnToggleAll.onClick = function()
local hayAlgunaMarcada = false
for i = 1, 24 do
if checkboxes[i].isChecked() then
hayAlgunaMarcada = true
break
end
end
if hayAlgunaMarcada then
for i = 1, 24 do
checkboxes[i].setChecked(false)
end
service.asyncSpeak("Todo desmarcado")
else
for i = 1, 24 do
checkboxes[i].setChecked(true)
end
service.asyncSpeak("Todo marcado")
end
actualizarBotonToggle()
end
btnCancel.onClick = function()
service.asyncSpeak("Configuración del rotor cancelada")
configDlg.dismiss()
end
btnSave.onClick = function()
local selectedTypes = {}
for i = 1, 24 do
if checkboxes[i].isChecked() then
selectedTypes[#selectedTypes + 1] = i - 1
end
end
if #selectedTypes > 0 then
saveRotorTypes(selectedTypes)
configDlg.dismiss()
service.reCreate()
task(100, function()
service.asyncSpeak("Configuración del rotor guardada correctamente")
end)
else
service.asyncSpeak("Necesitas al menos una casilla marcada para guardar cambios")
end
end
configDlg.setOnKeyListener(function(dialog, keyCode, event)
if keyCode == KeyEvent.KEYCODE_BACK and event.getAction() == KeyEvent.ACTION_UP then
service.asyncSpeak("Configuración del rotor cancelada")
configDlg.dismiss()
return true
end
return false
end)
configDlg.show()
end
showRotorConfigDialog()
return true
